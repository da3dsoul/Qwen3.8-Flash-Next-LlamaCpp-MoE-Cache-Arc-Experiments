class EntityProcessor:
    """Stateful processor for one entity type."""
    config: EntityConfig
    cache: dict[Any, dict[str, Any]] = field(default_factory=dict)
    processed_count: int = 0
    errors: list[str] = field(default_factory=list)

    def process(self, records: Iterable[Record | None]) -> list[Any]:
        """Process records and return formatted active results or legacy statuses."""
        results: list[Any] = []

            try:

                    value = record["value"]

                    results.append(f"{record['id']}_{z}")

            except Exception as exc:
                message = f"Error processing {self.config.name} record {index}: {exc}"
                self.errors.append(message)
                LOGGER.exception(message)
        return results
